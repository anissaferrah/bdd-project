Requete 3:
db.Voyage.aggregate([
  {
    $group: {
      _id: "$navette.moyen_transport.ligne.code",
      totalVoyages: { $sum: 1 }
    }
  },
  {
    $sort: { totalVoyages: -1 }
  },
  {
    $out: "Ligne-Voyages"
  }
])


#affichage de collection 
db["Ligne-Voyages"].find().pretty()
